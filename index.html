-- 1. Création de la table des profils CREATE TABLE public.profils ( id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY, email TEXT, role TEXT DEFAULT 'visiteur', -- 'visiteur', 'abonne', 'admin' date_fin_abonnement TIMESTAMPTZ, -- Date d'expiration created_at TIMESTAMPTZ DEFAULT NOW() ); -- 2. Activation de la sécurité (RLS) ALTER TABLE public.profils ENABLE ROW LEVEL SECURITY; -- 3. Les utilisateurs peuvent voir leur propre profil CREATE POLICY "Les utilisateurs voient leur profil" ON public.profils FOR SELECT TO authenticated USING (auth.uid() = id); -- 4. Un trigger pour créer automatiquement un profil quand quelqu'un s'inscrit CREATE OR REPLACE FUNCTION public.handle_new_user() RETURNS trigger AS $$ BEGIN INSERT INTO public.profils (id, email, role) VALUES (new.id, new.email, 'visiteur'); RETURN new; END; $$ LANGUAGE plpgsql SECURITY DEFINER; CREATE TRIGGER on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE PROCEDURE public.handle_new_user();
